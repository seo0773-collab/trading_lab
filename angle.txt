//@version=6
indicator("Reverse Protractor - Price/Bar Angle", overlay = true, max_lines_count = 200, max_labels_count = 100)

groupPoints = "Points"
float pivotPrice = input.price(0.0, "Pivot price", inline = "P", confirm = true, group = groupPoints)
int pivotTime    = input.time(0, "Pivot time", inline = "P", confirm = true, group = groupPoints)

float refPrice = input.price(0.0, "Reference price", inline = "R", confirm = true, group = groupPoints)
int refTime    = input.time(0, "Reference time", inline = "R", confirm = true, group = groupPoints)

groupDraw = "Draw"
int extendBars = input.int(100, "Extend bars each side", minval = 1, maxval = 500, group = groupDraw)
string drawMode = input.string("V from pivot", "Mode", options = ["V from pivot", "Straight through pivot", "Both"], group = groupDraw)
bool showOpposite = input.bool(true, "Show opposite mirror", group = groupDraw)
bool showLabels = input.bool(true, "Show labels", group = groupDraw)

groupScale = "Angle label"
float unit45 = input.float(1.0, "Price units per 1 bar at 45 deg", minval = 0.00000001, step = 0.1, group = groupScale)

var int pivotBar = na
var int refBar = na

if time == pivotTime
    pivotBar := bar_index

if time == refTime
    refBar := bar_index

var array<line> lines = array.new<line>()
var array<label> labels = array.new<label>()

clearDrawings() =>
    for id in lines
        line.delete(id)
    array.clear(lines)

    for id in labels
        label.delete(id)
    array.clear(labels)

addLine(int x1, float y1, int x2, float y2, color lineColor, int width, string styleName) =>
    line_style = styleName == "dotted" ? line.style_dotted : styleName == "dashed" ? line.style_dashed : line.style_solid
    id = line.new(x1, y1, x2, y2, xloc = xloc.bar_index, extend = extend.none, color = lineColor, width = width, style = line_style)
    array.push(lines, id)

addLabel(int x, float y, string txt, color bg) =>
    id = label.new(x, y, txt, xloc = xloc.bar_index, style = label.style_label_left, color = color.new(bg, 10), textcolor = color.white, size = size.small)
    array.push(labels, id)

if barstate.islast
    clearDrawings()

    bool ready = not na(pivotBar) and not na(refBar) and pivotBar != refBar

    if ready
        int leftX = pivotBar - extendBars
        int rightX = math.min(pivotBar + extendBars, bar_index + 500)

        float rawSlope = (refPrice - pivotPrice) / float(refBar - pivotBar)
        float slopeAbs = math.abs(rawSlope)
        float vDir = refPrice >= pivotPrice ? 1.0 : -1.0

        int barsLeft = pivotBar - leftX
        int barsRight = rightX - pivotBar

        float leftStraightY = pivotPrice - rawSlope * barsLeft
        float rightStraightY = pivotPrice + rawSlope * barsRight

        float leftVY = pivotPrice + vDir * slopeAbs * barsLeft
        float rightVY = pivotPrice + vDir * slopeAbs * barsRight

        float leftOppVY = pivotPrice - vDir * slopeAbs * barsLeft
        float rightOppVY = pivotPrice - vDir * slopeAbs * barsRight

        float leftOppStraightY = 2.0 * pivotPrice - leftStraightY
        float rightOppStraightY = 2.0 * pivotPrice - rightStraightY

        float normalizedAngle = math.atan(slopeAbs / unit45) * 180.0 / math.pi
        string info = "slope: " + str.tostring(slopeAbs, format.mintick) + "/bar\nangle: " + str.tostring(normalizedAngle, "#.##") + " deg"

        addLine(leftX, pivotPrice, rightX, pivotPrice, color.new(color.gray, 55), 1, "dotted")
        addLine(pivotBar, pivotPrice, refBar, refPrice, color.new(color.yellow, 0), 1, "dotted")

        if drawMode == "V from pivot" or drawMode == "Both"
            addLine(pivotBar, pivotPrice, rightX, rightVY, color.aqua, 2, "solid")
            addLine(pivotBar, pivotPrice, leftX, leftVY, color.aqua, 2, "solid")

            if showOpposite
                addLine(pivotBar, pivotPrice, rightX, rightOppVY, color.new(color.fuchsia, 0), 2, "solid")
                addLine(pivotBar, pivotPrice, leftX, leftOppVY, color.new(color.fuchsia, 0), 2, "solid")

        if drawMode == "Straight through pivot" or drawMode == "Both"
            addLine(leftX, leftStraightY, rightX, rightStraightY, color.lime, 2, "dashed")

            if showOpposite
                addLine(leftX, leftOppStraightY, rightX, rightOppStraightY, color.new(color.red, 0), 2, "dashed")

        if showLabels
            addLabel(pivotBar, pivotPrice, "Pivot\n" + info, color.blue)
            addLabel(refBar, refPrice, "Reference", color.orange)

    else if showLabels
        addLabel(bar_index, close, "Set Pivot and Reference on two different bars.", color.red)
